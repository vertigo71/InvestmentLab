# ============================================================
# InvestmentLab configuration help.
#
# Reference of all options available in the experiment
# files (*.yaml) of this folder. This file is not read by
# the program: it is documentation only.
#
# Each experiment file defines one complete experiment:
#
#   initial_capital     capital at the start of each window
#   cash_annual_rate    annual rate earned by uninvested cash
#   rolling:
#     window_years      length of each window, in years
#     start_date        "YYYY-MM-DD", quoted
#     end_date          "YYYY-MM-DD", quoted, excluded
#     data_source:      where prices come from (see below)
#     strategies:       list of strategies (see below)
#
# The experiment to run is chosen in active_yaml.txt.
# ============================================================

# ----------------------------------------------------------
# Data source
# ----------------------------------------------------------
#
# type: yahoo | csv
#
# ---- yahoo -----------------------------------------------
#
# ticker:
#   ^SP500TR       S&P 500 total return (dividends), 1990 -
#   ^GSPC          S&P 500 price only, no dividends
#   ^N225          Nikkei 225, in JPY, no dividends
#   URTH           iShares MSCI World ETF, 2012 -
#   VT             Vanguard Total World Stock ETF, 2008 -
#   0P00000RQC.F   European fund (Morningstar code)
#
# frequency (optional, default WEEKLY):
#   DAILY, WEEKLY, MONTHLY
#
# auto_adjust (optional, default true):
#   true    prices adjusted for dividends and splits
#   false   raw prices
#
# ---- csv -------------------------------------------------
#
# Reads prices from a local CSV file with two columns,
# date (YYYY-MM-DD) and close.
#
# label:
#   Short name used in output file names, e.g. SHILLER.
#
# path:
#   CSV file name, relative to the inputs folder.
#   shiller_sp_tr_monthly.csv   Shiller S&P Composite total
#                               return, monthly, 1871 -
#                               (from tools/shiller_converter)
#
# frequency (required, cannot be inferred from the file):
#   DAILY, WEEKLY, MONTHLY
#


# ----------------------------------------------------------
# Strategies
# ----------------------------------------------------------
#
# type: all_in
#   Invests all capital in the first period.
#   No other parameters.
#
# type: dca
#   Invests a fixed amount at a fixed frequency.
#   frequency: WEEKLY, BIWEEKLY, MONTHLY
#
# type: drawdown_dca
#   Like dca, multiplying the amount when the price
#   falls below a reference.
#   frequency:        WEEKLY, BIWEEKLY, MONTHLY
#   reference_price:  RUNNING_MAX  (highest close so far)
#                     RUNNING_AVG  (average close so far)
#   levels:           [[threshold, multiplier], ...]
#                     thresholds strictly increasing, < 1
#                     e.g. [[0.10, 2], [0.20, 3]]
#   base_multiplier:  (optional, default 1.0) applied when
#                     no threshold is reached
#
# Strategies whose frequency is not possible with the
# data (e.g. BIWEEKLY with MONTHLY data) are skipped.
#
